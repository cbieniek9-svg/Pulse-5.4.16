```javascript
function reopenInvestigation(db, id, actorName, opts = {}) {
  if (!opts.isManager) {
    const e = new Error('Manager login required to reopen an investigation.');
    e.status = 403; e.code = 'INVESTIGATION_REOPEN_FORBIDDEN'; throw e;
  }
  const row = getInvestigation(db, id);
  if (!row) { const e = new Error('Investigation not found.'); e.status = 404; throw e; }
  if (String(row.status) !== 'submitted') {
    const e = new Error('Only submitted investigations can be reopened.');
    e.status = 409; e.code = 'INVESTIGATION_LOCKED'; throw e;
  }
  db.run(
    `UPDATE incident_investigations SET status='draft', updated_at=?, updated_by=? WHERE id=?`,
    opts.serverTime || new Date().toISOString(), actorName, id,
  );
  appendAmendEvent(db, id, 'reopen', actorName, opts.serverTime || new Date().toISOString());
  return getInvestigation(db, id);
}
```

Route: `POST /api/safety/investigations/:id/reopen` with manager check + `logManagerAudit(... action: 'incident_investigation_reopened')`.
